{{- /* Returns paper buttons ({name, url}) for a software package, read from related_papers (publication folder names) plus legacy related_paper. Each button opens the paper itself: DOI, else preprint, else full text, else the paper's page on this site. Unknown slugs are skipped. With more than one paper, the label names the venue, e.g. "Paper (The R Journal)". */ -}} {{- $slugs := .Params.related_papers | default slice -}} {{- with .Params.related_paper -}}{{- $slugs = $slugs | append . -}}{{- end -}} {{- $pubs := slice -}} {{- range $s := $slugs -}} {{- with site.GetPage (printf "/publication/%s" $s) -}}{{- $pubs = $pubs | append . -}}{{- end -}} {{- end -}} {{- $out := slice -}} {{- range $pubs -}} {{- $url := .RelPermalink -}} {{- with .Params.url_source -}}{{- $url = . -}}{{- end -}} {{- with .Params.url_preprint -}}{{- $url = . -}}{{- end -}} {{- with .Params.doi -}}{{- $url = printf "https://doi.org/%s" . -}}{{- end -}} {{- $name := "Paper" -}} {{- if gt (len $pubs) 1 -}}{{- $name = printf "Paper (%s)" (.Params.journal | default .Date.Year) -}}{{- end -}} {{- $out = $out | append (dict "name" $name "url" (string $url)) -}} {{- end -}} {{- return $out -}}
